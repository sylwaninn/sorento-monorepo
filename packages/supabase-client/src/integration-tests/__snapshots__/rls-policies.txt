profiles :: profiles_select
  command    SELECT (permissive)
  roles      {authenticated}
  using      ((id = auth.uid()) OR (EXISTS ( SELECT 1
   FROM (memberships m1
     JOIN memberships m2 ON ((m2.dossier_id = m1.dossier_id)))
  WHERE ((m1.user_id = auth.uid()) AND (m2.user_id = profiles.id) AND (m1.role = ANY (ARRAY['viewer'::text, 'collaborator'::text, 'owner'::text])) AND (m2.role = ANY (ARRAY['viewer'::text, 'collaborator'::text, 'owner'::text]))))))
  with check -
